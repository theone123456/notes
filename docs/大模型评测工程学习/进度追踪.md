const STORE_KEY = "eval-plan-tracker-v1";

// ===== 状态读写 =====
let state = { done: {}, logs: [] };

function load() {
  try {
    const raw = localStorage.getItem(STORE_KEY);
    if (raw) {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") {
        state.done = parsed.done || {};
        state.logs = Array.isArray(parsed.logs) ? parsed.logs : [];
      }
    }
  } catch (e) { console.warn("读取进度失败,使用空状态", e); }
}

function save() {
  localStorage.setItem(STORE_KEY, JSON.stringify(state));
}

// ===== 日期工具(本地时区,避免 toISOString 的 UTC 偏移问题)=====
function pad(n) { return String(n).padStart(2, "0"); }
function d2str(d) { return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`; }
function todayStr() { return d2str(new Date()); }
function str2d(s) { const [y, m, d] = s.split("-").map(Number); return new Date(y, m - 1, d); }
function daysBetween(fromStr, toStr) {
  return Math.round((str2d(toStr) - str2d(fromStr)) / 86400000);
}

// ===== 连续记录天数:从今天(或昨天)往前数连续有记录的天数 =====
function calcStreak() {
  const days = new Set(state.logs.map(l => l.d));
  if (days.size === 0) return 0;
  let cursor = new Date();
  if (!days.has(d2str(cursor))) {
    // 今天还没记不算断,从昨天开始算
    cursor.setDate(cursor.getDate() - 1);
    if (!days.has(d2str(cursor))) return 0;
  }
  let streak = 0;
  while (days.has(d2str(cursor))) {
    streak++;
    cursor.setDate(cursor.getDate() - 1);
  }
  return streak;
}

function renderStats() {
  const doneCount = MILESTONES.filter(m => state.done[m.id]).length;
  document.getElementById("stat-overall").textContent = `${doneCount} / ${MILESTONES.length}`;

  // 下一里程碑:第一个未完成项,计算剩余天数(负数=已逾期)
  const next = MILESTONES.find(m => !state.done[m.id]);
  const el = document.getElementById("stat-next");
  if (!next) {
    el.textContent = "全部完成";
    el.className = "value ok";
  } else {
    const diff = daysBetween(todayStr(), next.date);
    el.textContent = diff >= 0 ? `还有 ${diff} 天` : `逾期 ${-diff} 天`;
    el.className = "value " + (diff < 0 ? "warn" : "accent");
  }

  const streak = calcStreak();
  const sEl = document.getElementById("stat-streak");
  sEl.textContent = `${streak} 天`;
  sEl.className = "value " + (streak >= 7 ? "ok" : streak >= 1 ? "accent" : "");

  const tEl = document.getElementById("stat-today");
  const loggedToday = state.logs.some(l => l.d === todayStr());
  tEl.textContent = loggedToday ? "已记录" : "未记录";
  tEl.className = "value " + (loggedToday ? "ok" : "warn");
}

function renderPhases() {
  const box = document.getElementById("phases");
  box.innerHTML = "";
  PHASES.forEach(p => {
    const total = p.ms.length;
    const done = p.ms.filter(id => state.done[id]).length;
    const pct = total ? Math.round(done / total * 100) : 0;

    const card = document.createElement("div");
    card.className = "phase-card";
    card.innerHTML = `
      <div class="phase-head">
        <div>
          <span class="phase-name">${p.name}</span>
          <span class="phase-range">(${p.range})</span>
        </div>
        <span class="phase-count">${done} / ${total} · ${pct}%</span>
      </div>
      <div class="bar"><div style="width:${pct}%"></div></div>`;
    box.appendChild(card);
  });
}

function renderMilestones() {
  const box = document.getElementById("milestones");
  box.innerHTML = "";
  const today = todayStr();

  MILESTONES.forEach(m => {
    const isDone = !!state.done[m.id];
    const diff = daysBetween(today, m.date);

    const item = document.createElement("div");
    item.className = "ms-item" + (isDone ? " done" : diff < 0 ? " overdue" : "");

    const statusHtml = isDone
      ? ""
      : diff < 0
        ? `<span class="ms-status overdue">逾期 ${-diff} 天</span>`
        : diff <= 3
          ? `<span class="ms-status soon">即将到期</span>`
          : "";

    const stamp = isDone ? `<div class="ms-stamp">完成于 ${state.done[m.id]}</div>` : "";

    item.innerHTML = `
      <input type="checkbox" class="ms-check" data-id="${m.id}" ${isDone ? "checked" : ""}>
      <div class="ms-body">
        <span class="ms-date">${m.date}</span>
        <div class="ms-text">${m.text}</div>
        ${stamp}
      </div>
      ${statusHtml}`;
    box.appendChild(item);
  });
}

function renderLogs() {
  const box = document.getElementById("log-list");
  box.innerHTML = "";
  const sorted = [...state.logs].sort((a, b) => (a.d < b.d ? 1 : -1));

  if (sorted.length === 0) {
    box.innerHTML = `<div class="empty">还没有日志。每天收尾 10 分钟,今日做了什么 / 卡在哪 / 明天做什么。</div>`;
    return;
  }
  // 只展示最近 60 条,避免列表过长
  sorted.slice(0, 60).forEach((l, i) => {
    const row = document.createElement("div");
    row.className = "log-item";
    row.innerHTML = `
      <span class="d">${l.d}</span>
      <span class="t"></span>
      <button class="del" data-i="${i}">删除</button>`;
    row.querySelector(".t").textContent = l.t; // 用 textContent 防注入
    box.appendChild(row);
  });
}

function renderAll() {
  renderStats();
  renderPhases();
  renderMilestones();
  renderLogs();
}

// ===== 交互事件(事件委托)=====
document.getElementById("milestones").addEventListener("change", e => {
  if (!e.target.matches(".ms-check")) return;
  const id = e.target.dataset.id;
  if (e.target.checked) {
    state.done[id] = todayStr(); // 打卡时间 = 今天
  } else {
    delete state.done[id];
  }

  renderAll();
});

document.getElementById("log-add").addEventListener("click", () => {
  const d = document.getElementById("log-date").value || todayStr();
  const t = document.getElementById("log-text").value.trim();
  if (!t) { alert("写点什么再记录吧(一行即可)"); return; }
  // 同一天重复记录则覆盖,保持一天一条
  state.logs = state.logs.filter(l => l.d !== d);
  state.logs.push({ d, t });
  save();
  document.getElementById("log-text").value = "";
  renderAll();
});

  if (e.key === "Enter") document.getElementById("log-add").click();

document.getElementById("log-list").addEventListener("click", e => {
  if (!e.target.matches(".del")) return;
  const idx = Number(e.target.dataset.i);
  // 注意索引基于排序后的展示列表,需映射回原数组
  const sorted = [...state.logs].sort((a, b) => (a.d < b.d ? 1 : -1));
  const target = sorted[idx];
  if (target && confirm(`删除 ${target.d} 的日志?`)) {
    state.logs = state.logs.filter(l => !(l.d === target.d && l.t === target.t));
    save();
    renderAll();
  }
});

document.getElementById("btn-import").addEventListener("click", () => {
  document.getElementById("file-import").click();

document.getElementById("file-import").addEventListener("change", e => {
  const file = e.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = () => {
    try {
      const data = JSON.parse(reader.result);
      if (!data || typeof data !== "object" || !Array.isArray(data.logs) || typeof data.done !== "object") {
        throw new Error("格式不符");
      }
      if (!confirm(`导入将覆盖当前数据(含 ${data.logs.length} 条日志)。继续?`)) return;
      state = { done: data.done, logs: data.logs };
      save();
      renderAll();
    } catch (err) {
      alert(`导入失败:${err.message}`);
    } finally {
      e.target.value = "";
    }
  };
  reader.readAsText(file);
});

document.getElementById("btn-reset").addEventListener("click", () => {
